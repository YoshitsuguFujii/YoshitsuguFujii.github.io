# andで条件組み立て
def condition
  cons = []
  cons << User.arel_table[:name].matches("%hoge%") 
  cons << User.arel_table[:nick_name].matches("%fuga%")
  cons.inject(&:and)
end

user.where(condition)

配列の結合で一瞬悩んだのでメモ。
andをorにすると、どれかに一致するという条件になります。
